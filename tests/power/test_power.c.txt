void bsp_gpio_write(GPIO_TypeDef *port, uint16_t pin, GPIO_PinState value)
{
    (void)port;
    unsigned id=99;
    switch(pin) {
    case GPIO_PIN_0:id=RELAY_INV_OUT;break;
    case GPIO_PIN_1:id=RELAY_AC_CTRL;break;
    case GPIO_PIN_2:id=RELAY_CHG_IN;break;
    case GPIO_PIN_3:id=RELAY_CHG_OUT;break;
    case GPIO_PIN_4:id=RELAY_INV_PRO;break;
    case GPIO_PIN_7:id=RELAY_INV_IN;break;
    }
    CHECK(id<RELAY_CNT);
    if (id>=RELAY_CNT) return;
    if(value) gpio_bits|=BIT(id); else gpio_bits&=~BIT(id);
    ++gpio_writes;
    CHECK(!(gpio_bits&BIT(RELAY_INV_OUT)) || ((gpio_bits&MAIN_BITS)==MAIN_BITS));
    if(value && id==RELAY_INV_IN) CHECK(gpio_bits&BIT(RELAY_INV_PRO));
    if(!value && id==RELAY_INV_PRO) CHECK(!(gpio_bits&BIT(RELAY_INV_OUT)));
    if(!value && id==RELAY_INV_IN) CHECK(!(gpio_bits&BIT(RELAY_INV_PRO)));
    CHECK(!((gpio_bits&ALL_INV_BITS) && (gpio_bits&(CHARGE_BITS|BYPASS_BITS))));
}

GPIO_PinState bsp_gpio_read(GPIO_TypeDef *port, uint16_t pin)
{
    for (unsigned i=0;i<KEY_IDX_CNT;i++)
        if(s_pin_map[i].port==port && s_pin_map[i].pin==pin)
            return (pressed_keys&s_pin_map[i].key_value) ? GPIO_PIN_RESET : GPIO_PIN_SET;
    return GPIO_PIN_SET;
}
static void record_key(uint8_t key)
{
    ++key_events;last_key_event=key;
    if(key&KEY_LONG_MARK)++long_events;
}
static void sample(float v) { ac_input_update(true,v); }
static void reset(bool ac)
{
    now_ms += 10000;
    s_state=SYS_STATUS_IDLE;s_shutting_down=false;s_shutdown_complete=false;
    s_alarm=0;power_abort();s_dc_release_ms=PATH_RELEASE_MS;
    s_ac_input=(ac_input_snapshot_t){0};s_ac_good_samples=0;
    test_temp=(TEMP_INFO_T){0};battery_good=true;battery_faults=0;
    display_ready=true;s_ui_dirty=true;power_off_count=checkpoint_count=0;
    if(ac) {sample(220);sample(220);} else sample(0);
}
static void tick(uint32_t ms) { now_ms+=ms;work_status_process();power_process();display_publish(); }
static void finish_invert(void)
{
    tick(PATH_RELEASE_MS);tick(PATH_RELEASE_MS);
    CHECK(gpio_bits==PRECHARGE_BITS);
    tick(PATH_PRECHARGE_MS);CHECK(gpio_bits==MAIN_BITS);
    tick(PATH_INVERTER_START_MS);CHECK(gpio_bits==INVERT_BITS);
    CHECK(s_state==SYS_STATUS_OUTPOWER);
}
static void start_invert(void) { sample(0);handle_path_key(false);finish_invert(); }
static void start_charge(void)
{
    sample(220);sample(220);handle_path_key(true);
    tick(PATH_RELEASE_MS);tick(PATH_RELEASE_MS);tick(PATH_CHARGER_START_MS);
    CHECK(gpio_bits==CHARGE_BITS);CHECK(s_state==SYS_STATUS_CHARGE);
}
static void stop_settle(void) { tick(PATH_RELEASE_MS);tick(PATH_CONTACTOR_RELEASE_MS); }
static void to_inverter_stage(unsigned stage)
{
    reset(false);handle_path_key(false);
    if(stage>=1)tick(PATH_RELEASE_MS);
    if(stage>=2)tick(PATH_RELEASE_MS);
    if(stage>=3)tick(PATH_PRECHARGE_MS);
    if(stage>=4)tick(PATH_INVERTER_START_MS);
}
__attribute__((noinline)) void test_done(void) { for(;;) {} }
void test_main(void)
{
    reset(false);sample(90);CHECK(mod_metering_get_ac_input().state==AC_INPUT_UNKNOWN);
    sample(90);CHECK(mod_metering_get_ac_input().state==AC_INPUT_PRESENT);
    sample(300);CHECK(mod_metering_get_ac_input().state==AC_INPUT_PRESENT);
    sample(300.1f);CHECK(mod_metering_get_ac_input().state==AC_INPUT_ABSENT);
    sample(89.9f);CHECK(mod_metering_get_ac_input().state==AC_INPUT_ABSENT);
    sample(220);sample(220);now_ms+=2501;CHECK(mod_metering_get_ac_input().state==AC_INPUT_UNKNOWN);
    sample(220);CHECK(mod_metering_get_ac_input().state==AC_INPUT_UNKNOWN);
    sample(220);ac_input_update(false,0);CHECK(mod_metering_get_ac_input().state==AC_INPUT_UNKNOWN);
    sample(0.0f/0.0f);CHECK(mod_metering_get_ac_input().state==AC_INPUT_UNKNOWN);
    sample(1.0f/0.0f);CHECK(mod_metering_get_ac_input().state==AC_INPUT_UNKNOWN);
    reset(true);input_good=false;output_good=true;output_voltage=220;
    mod_metering_handler();CHECK(mod_metering_get_ac_input().state==AC_INPUT_UNKNOWN);
    input_good=true;input_voltage=220;now_ms+=1000;mod_metering_handler();
    CHECK(mod_metering_get_ac_input().state==AC_INPUT_UNKNOWN);
    now_ms+=1000;mod_metering_handler();CHECK(mod_metering_get_ac_input().state==AC_INPUT_PRESENT);
    CHECK(s_info.vin_freq_hz==50 && s_info.vout_freq_hz==50);
    input_good=false;
    for(unsigned i=0;i<10;i++){now_ms+=1000;mod_metering_handler();}
    CHECK(s_info.vin_v==0 && s_info.vin_freq_hz==0 && s_info.vout_freq_hz==50);
    CHECK(mod_metering_get_ac_input().state==AC_INPUT_UNKNOWN);input_good=true;

    /* Both DC pins required, accurate stabilization and ordered shutdown. */
    reset(false);CHECK(!drv_relay_on(RELAY_INV_OUT));
    CHECK(!drv_relay_on(RELAY_INV_IN));CHECK(drv_relay_on(RELAY_INV_PRO));CHECK(!drv_relay_on(RELAY_INV_OUT));
    drv_relay_all_off();
    handle_path_key(false);tick(49);CHECK(gpio_bits==0);tick(1);tick(50);
    CHECK(gpio_bits==PRECHARGE_BITS);
    tick(499);CHECK(gpio_bits==PRECHARGE_BITS);tick(1);CHECK(gpio_bits==MAIN_BITS);
    tick(99);CHECK(gpio_bits==MAIN_BITS);tick(1);CHECK(gpio_bits==INVERT_BITS);
    CHECK(!drv_relay_on(RELAY_AC_CTRL));CHECK(!drv_relay_on(RELAY_CHG_IN));
    unsigned writes=gpio_writes;_key_event_handler(KEY_LONG_VALUE_ACOUT);CHECK(writes==gpio_writes);
    handle_path_key(false);CHECK(gpio_bits==0 && !s_output_requested);
    tick(49);CHECK(gpio_bits==0);tick(1);CHECK(gpio_bits==0);
    tick(49);CHECK(s_phase==PHASE_RELEASE_DC);tick(1);CHECK(s_state==SYS_STATUS_IDLE);
    start_invert();

    /* Automatic bypass keeps main DC on until AC output release completes. */
    sample(220);sample(220);power_process();CHECK(gpio_bits==0);
    tick(49);CHECK(gpio_bits==0);tick(1);CHECK(gpio_bits==0);
    tick(49);CHECK(gpio_bits==0);tick(1);CHECK(gpio_bits==BYPASS_BITS);
    CHECK(s_state==SYS_STATUS_BYPASS && !s_charge_requested);
    sample(0);tick(1);CHECK(gpio_bits==0 && !s_output_requested);
    tick(1000);CHECK(gpio_bits==0);start_invert();
    sample(220);sample(220);power_process();sample(301);tick(1);CHECK(gpio_bits==0);
    tick(1000);CHECK(gpio_bits==0);

    /* Every startup/running phase: off, fault, shutdown and AC arrival. */
    for(unsigned stage=0;stage<5;stage++) {
        to_inverter_stage(stage);handle_path_key(false);stop_settle();tick(1200);
        CHECK(gpio_bits==0 && !s_output_requested);start_invert();
        to_inverter_stage(stage);enter_fault(ALM_INV_OTP);tick(1200);
        CHECK(gpio_bits==0 && s_state==SYS_STATUS_FAULT);
        handle_path_key(false);CHECK(gpio_bits==0);
        to_inverter_stage(stage);sample(220);sample(220);power_process();stop_settle();
        CHECK(gpio_bits==BYPASS_BITS && s_output_requested && !s_charge_requested);
    }
    /* UNKNOWN stalls each inverter start phase and then aborts; never closes PD0. */
    for(unsigned stage=2;stage<4;stage++) {
        to_inverter_stage(stage);ac_input_update(false,0);tick(600);
        CHECK(!(gpio_bits&BIT(RELAY_INV_OUT)));tick(PATH_INPUT_WAIT_MS);
        CHECK(gpio_bits==0 && s_target_path==PATH_NONE);
    }
    reset(true);handle_path_key(false);stop_settle();CHECK(gpio_bits==BYPASS_BITS);
    CHECK(!drv_relay_on(RELAY_INV_PRO));CHECK(!drv_relay_on(RELAY_INV_IN));CHECK(!drv_relay_on(RELAY_INV_OUT));
    handle_path_key(false);stop_settle();CHECK(gpio_bits==0);
    handle_path_key(false);stop_settle();CHECK(gpio_bits==BYPASS_BITS);

    reset(true);start_charge();writes=gpio_writes;handle_path_key(false);
    CHECK(gpio_bits==(CHARGE_BITS|BYPASS_BITS) && gpio_writes==writes+1);
    CHECK(s_state==SYS_STATUS_CHARGE_BYPASS);display_publish();CHECK(icon1==1 && icon2==2);
    handle_path_key(false);CHECK(gpio_bits==CHARGE_BITS && !s_output_requested);
    handle_path_key(false);handle_path_key(true);
    CHECK(gpio_bits==(BYPASS_BITS|BIT(RELAY_CHG_IN)));
    tick(PATH_RELEASE_MS);CHECK(gpio_bits==BYPASS_BITS && !s_charge_requested);
    sample(220);sample(220);handle_path_key(true);tick(50);tick(50);tick(1500);
    CHECK(gpio_bits==CHARGE_BITS && !s_output_requested);
    writes=gpio_writes;_key_event_handler(KEY_LONG_VALUE_ACIN);CHECK(writes==gpio_writes);
    handle_path_key(true);stop_settle();CHECK(gpio_bits==0);start_charge();
    handle_path_key(false);ac_input_update(false,0);tick(1);CHECK(gpio_bits==0);
    display_publish();CHECK(icon1==0 && icon2==0);
    handle_path_key(false);CHECK(s_target_path==PATH_NONE);
    reset(false);handle_path_key(true);CHECK(gpio_bits==0 && buzzer_count==3 && !s_charge_requested);
    reset(false);ac_input_update(false,0);handle_path_key(true);CHECK(buzzer_count==3 && !s_charge_requested);
    reset(false);start_invert();writes=gpio_writes;handle_path_key(true);
    CHECK(gpio_bits==INVERT_BITS && writes==gpio_writes && buzzer_count==3);

    reset(true);handle_path_key(true);tick(50);tick(50);handle_path_key(true);stop_settle();
    tick(1500);CHECK(gpio_bits==0);start_charge();
    reset(true);start_charge();handle_path_key(false);test_temp.chg_temp=BOARD_PROT_CHG_TMP_FAULT_TH;
    tick(1);CHECK(gpio_bits==0 && s_state==SYS_STATUS_FAULT);
    for(unsigned stage=2;stage<4;stage++) {
        to_inverter_stage(stage);battery_good=false;tick(1);
        CHECK(gpio_bits==0 && s_state==SYS_STATUS_FAULT);
        to_inverter_stage(stage);test_temp.inv_temp=BOARD_PROT_INV_TMP_FAULT_TH;tick(1);
        CHECK(gpio_bits==0 && s_state==SYS_STATUS_FAULT);
    }
    reset(false);battery_faults=BAT_FAULT_UV;handle_path_key(false);CHECK(gpio_bits==0);
    reset(true);battery_faults=BAT_FAULT_UV;start_charge();
    reset(true);start_charge();sample(0);tick(1);CHECK(gpio_bits==0);
    reset(true);handle_path_key(false);stop_settle();tick(2501);CHECK(gpio_bits==0);
    reset(false);start_invert();display_ready=false;enter_fault(ALM_INV_OTP);display_publish();
    CHECK(s_ui_dirty);display_ready=true;display_publish();CHECK(!s_ui_dirty && icon1==0 && icon2==0);
    /* Existing immediate board power-off is outside this relay sequence test. */
    reset(false);start_invert();ac_input_update(false,0);handle_path_key(false);stop_settle();CHECK(gpio_bits==0);

    /* Actual key scanner: ONOFF 3000ms, early release, no repeated long event. */
    reset(false);mod_key_init();mod_key_register_callback(record_key);
    pressed_keys=KEY_VALUE_ONOFF;key_events=long_events=0;mod_key_poll_handler();
    now_ms+=20;mod_key_poll_handler();now_ms+=20;mod_key_poll_handler();
    CHECK(key_events==1 && last_key_event==KEY_VALUE_ONOFF);
    now_ms+=2959;mod_key_poll_handler();CHECK(long_events==0);
    now_ms+=1;mod_key_poll_handler();CHECK(long_events==1 && last_key_event==(KEY_LONG_VALUE_ONOFF));
    now_ms+=3000;mod_key_poll_handler();CHECK(long_events==1);
    pressed_keys=0;mod_key_poll_handler();pressed_keys=KEY_VALUE_ONOFF;mod_key_poll_handler();
    now_ms+=20;mod_key_poll_handler();now_ms+=20;mod_key_poll_handler();
    now_ms+=2959;pressed_keys=0;mod_key_poll_handler();CHECK(long_events==1);
    now_ms+=100;pressed_keys=KEY_VALUE_ONOFF;mod_key_poll_handler();
    now_ms+=20;mod_key_poll_handler();now_ms+=20;mod_key_poll_handler();
    now_ms+=760;mod_key_poll_handler();CHECK(long_events==1);
    pressed_keys=0;mod_key_poll_handler();
    /* Clock rollover also retains 3 second threshold. */
    now_ms=0xfffffff0u;pressed_keys=KEY_VALUE_ONOFF;mod_key_poll_handler();
    now_ms+=20;mod_key_poll_handler();now_ms+=20;mod_key_poll_handler();
    now_ms+=2959;mod_key_poll_handler();CHECK(long_events==1);
    now_ms++;mod_key_poll_handler();CHECK(long_events==2);
    reset(false);now_ms=0xffffffd0u;sample(0);handle_path_key(false);finish_invert();
    test_done();
}
